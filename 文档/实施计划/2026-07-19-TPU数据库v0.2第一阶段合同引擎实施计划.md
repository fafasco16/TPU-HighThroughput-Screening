```python
@dataclass(frozen=True)
class ContractBundle:
    schema_version: str
    schema: dict[str, Any]
    enums: dict[str, Any]
    rules: dict[str, Any]
    document_hashes: dict[str, str]


class ContractValidationError(ValueError):
    pass


def validate_contract_bundle(
    schema: Mapping[str, Any],
    enums: Mapping[str, Any],
    rules: Mapping[str, Any],
) -> None:
    validate_schema_definition(schema, enums)
    versions = {schema.get("schema_version"), enums.get("schema_version"), rules.get("schema_version")}
    if len(versions) != 1 or None in versions:
        raise ContractValidationError("schema, enums and rules require one schema_version")
    _validate_table_constraints(schema["tables"])
    _validate_conditional_required(schema)
    _validate_rule_catalog(rules)


def load_contract_bundle(schema_path, enum_path, rule_path) -> ContractBundle:
    documents = {
        "schema": _load_yaml_mapping(schema_path),
        "enums": _load_yaml_mapping(enum_path),
        "rules": _load_yaml_mapping(rule_path),
    }
    validate_contract_bundle(documents["schema"], documents["enums"], documents["rules"])
    hashes = {
        name: hashlib.sha256(
            json.dumps(document, ensure_ascii=False, sort_keys=True, separators=(",", ":")).encode("utf-8")
        ).hexdigest()
        for name, document in documents.items()
    }
    return ContractBundle(
        schema_version=documents["schema"]["schema_version"],
        schema=documents["schema"],
        enums=documents["enums"],
        rules=documents["rules"],
        document_hashes=hashes,
    )
```

约束验证必须检查：名称非空且在同表唯一；fields 非空、无重复且均存在；FK 本地/目标列数相等且目标表/字段存在；`on_delete` 只允许 `restrict/cascade/set_null/no_action`；CHECK 有非空 expression；conditional rule 的 table、when field 和 require fields 均存在。

```python
ROOT_NAMESPACE = UUID("1b3452dd-f305-5c9e-b55a-4f782ea67d10")


def canonical_identity_json(value: object) -> str:
    normalized = _normalize(value)
    return json.dumps(normalized, ensure_ascii=False, sort_keys=True, separators=(",", ":"), allow_nan=False)


def identity_key_sha256(identity_key: object) -> str:
    return hashlib.sha256(canonical_identity_json(identity_key).encode("utf-8")).hexdigest()


def stable_record_uid(entity_type: str, identity_key: object, *, algorithm_version: str = "uuid5-v1") -> str:
    entity_type = _token(entity_type, "entity_type")
    algorithm_version = _token(algorithm_version, "algorithm_version")
    namespace = uuid5(ROOT_NAMESPACE, f"{algorithm_version}:{entity_type}")
    return str(uuid5(namespace, canonical_identity_json(identity_key)))


def content_sha256(content: object) -> str:
    return hashlib.sha256(canonical_identity_json(content).encode("utf-8")).hexdigest()


def stable_revision_id(record_uid: str, schema_version: str, content: object) -> str:
    record_uuid = UUID(record_uid)
    return str(uuid5(record_uuid, f"{_token(schema_version, 'schema_version')}:{content_sha256(content)}"))
```

`代码/logical_hash.py` 固定 `ALGORITHM_VERSION = "tpu-logical-hash/1"`，并实现：

```python
def _frame(payload: bytes) -> bytes:
    return len(payload).to_bytes(8, "big") + payload


def _canonical_json_bytes(value: object) -> bytes:
    normalized = json.loads(canonical_identity_json(value))
    return json.dumps(normalized, ensure_ascii=False, sort_keys=True, separators=(",", ":"), allow_nan=False).encode("utf-8")


def canonical_typed_value(value: object, type_name: str) -> bytes:
    supported = {"string", "integer", "float64", "boolean", "date", "datetime", "json", "binary", "missing"}
    if type_name not in supported:
        raise LogicalHashError(f"unsupported logical type: {type_name}")
    if value is None:
        return b"N"
    if type_name == "string":
        if not isinstance(value, str):
            raise LogicalHashError("string value required")
        return b"S" + _frame(unicodedata.normalize("NFC", value).encode("utf-8"))
    if type_name == "integer":
        if not isinstance(value, int) or isinstance(value, bool):
            raise LogicalHashError("integer value required")
        return b"I" + _frame(str(value).encode("ascii"))
    if type_name == "float64":
        numeric = float(value)
        payload = bytes.fromhex("7ff8000000000000") if math.isnan(numeric) else struct.pack(">d", numeric)
        return b"F" + _frame(payload)
    if type_name == "boolean":
        if not isinstance(value, bool):
            raise LogicalHashError("boolean value required")
        return b"B1" if value else b"B0"
    if type_name == "date":
        parsed = date.fromisoformat(value) if isinstance(value, str) else value
        if not isinstance(parsed, date) or isinstance(parsed, datetime):
            raise LogicalHashError("date value required")
        return b"D" + _frame(parsed.isoformat().encode("ascii"))
    if type_name == "datetime":
        parsed = datetime.fromisoformat(value.replace("Z", "+00:00")) if isinstance(value, str) else value
        if not isinstance(parsed, datetime) or parsed.tzinfo is None:
            raise LogicalHashError("timezone-aware datetime required")
        utc_text = parsed.astimezone(timezone.utc).isoformat(timespec="microseconds").replace("+00:00", "Z")
        return b"T" + _frame(utc_text.encode("ascii"))
    if type_name == "binary":
        if not isinstance(value, bytes):
            raise LogicalHashError("bytes value required")
        return b"X" + _frame(base64.b64encode(value))
    if type_name == "missing":
        if not isinstance(value, Mapping) or set(value) != {"missing_reason"} or not isinstance(value["missing_reason"], str):
            raise LogicalHashError("missing value requires one missing_reason")
        return b"M" + _frame(value["missing_reason"].encode("utf-8"))
    return b"J" + _frame(_canonical_json_bytes(value))


def row_logical_hash(row: Mapping[str, object], schema: Sequence[tuple[str, str]]) -> str:
    payload = b"".join(_frame(name.encode("utf-8")) + _frame(canonical_typed_value(row.get(name), type_name)) for name, type_name in schema)
    return hashlib.sha256(payload).hexdigest()


def table_logical_hash(frame: pd.DataFrame, *, schema: Sequence[tuple[str, str]], sort_key: Sequence[str]) -> str:
    missing = [name for name, _ in schema if name not in frame.columns]
    if missing:
        raise LogicalHashError(f"missing schema columns: {', '.join(missing)}")
    ordered = frame.sort_values(list(sort_key), kind="mergesort", na_position="first")
    schema_hash = hashlib.sha256(json.dumps(list(schema), separators=(",", ":")).encode()).digest()
    row_hashes = b"".join(bytes.fromhex(row_logical_hash(row, schema)) for row in ordered.to_dict("records"))
    return hashlib.sha256(ALGORITHM_VERSION.encode() + schema_hash + row_hashes).hexdigest()


def snapshot_logical_hash(tables: Mapping[str, tuple[int, str]]) -> str:
    canonical = json.dumps(sorted((name, count, digest) for name, (count, digest) in tables.items()), separators=(",", ":"))
    return hashlib.sha256((ALGORITHM_VERSION + canonical).encode()).hexdigest()
```

模块导入 `base64/hashlib/json/math/struct/unicodedata`、`date/datetime/timezone`、`Mapping/Sequence` 和 `canonical_identity_json`；支持类型限定为 `string/integer/float64/boolean/date/datetime/json/binary/missing`，未知类型立即失败。
